// === БЛОК 1: ИНИЦИАЛИЗАЦИЯ ПРИ ПУСКЕ ===
000000 LD M8002           // При первом скане после подачи питания
000001 RST Y0             // Выключить вентилятор
000002 RST Y1             // Выключить ККБ (охлаждение)
000003 RST Y2             // Закрыть жалюзи
000004 RST Y3             // Выключить лампу "Авария"
000005 RST Y4             // Выключить лампу "Замерз"
000006 RST Y6             // Выключить нагрев
000007 RST M512           // Сбросить флаг работы системы
000008 RST M22            // Сбросить флаг аварии

// === БЛОК 2: СИСТЕМА ПУСК/СТОП (из старого кода) ===
000009 LD X0              // Если нажата кнопка ПУСК
000010 OR M512            // ИЛИ система уже работает
000011 AND X1             // И кнопка СТОП не нажата (нормально замкнута)
000012 ANI M22            // И нет аварийных сигналов
000013 OUT M512           // Тогда система ВКЛЮЧЕНА

000014 LD X1              // Если кнопка СТОП
000015 ANI X1             // (когда нажата - нормально замкнутый контакт разомкнут)
000016 RST M512           // Тогда система ВЫКЛЮЧЕНА
000017 OUT M100 ==== ИСПРАВЛЕНИЕ!!!

// === БЛОК 3: ПРОВЕРКА АВАРИЙНЫХ СИГНАЛОВ ===
000017 LD X7              // Авария от температурного контроллера DT3
000018 OR X5              // ИЛИ сигнал замерзания
000019 OUT M22            // Установить флаг аварии

// === БЛОК 4: ВЫБОР РЕЖИМА РАБОТЫ ПО СЕЛЕКТОРУ ===
000020 LD M512            // Если система включена
000021 ANI M22            // И нет аварий
000022 ANI X2             // И не выбран нагрев
000023 ANI X3             // И не выбрано охлаждение
000024 OUT M2             // Активировать режим ВЕНТИЛЯЦИИ

000025 LD M512            // Если система включена
000026 ANI M22            // И нет аварий
000027 AND X2             // И выбран нагрев
000028 ANI X3             // И не выбрано охлаждение
000029 OUT M3             // Активировать режим НАГРЕВА

000030 LD M512            // Если система включена
000031 ANI M22            // И нет аварий
000032 AND X3             // И выбрано охлаждение
000033 ANI X2             // И не выбран нагрев
000034 OUT M4             // Активировать режим ОХЛАЖДЕНИЯ

// === БЛОК 5: РЕЖИМ ВЕНТИЛЯЦИИ ===
000035 LD M2              // Если активен режим вентиляции
000036 ANI M22            // И нет аварий
000037 RST Y1             // Выключить ККБ (охлаждение)
000038 RST Y6             // Выключить нагрев
000039 OUT Y2             // ОТКРЫТЬ жалюзи
000040 TMR T10 K300       // Запустить таймер 30 сек на открытие жалюзи

000041 LD T10             // После истечения таймера T10 (30 сек)
000042 OUT Y0             // ВКЛЮЧИТЬ вентилятор

// === БЛОК 6: РЕЖИМ НАГРЕВА ===
000043 LD M3              // Если активен режим нагрева
000044 ANI M22            // И нет аварий
000045 RST Y1             // Выключить ККБ (охлаждение)
000046 OUT Y2             // ОТКРЫТЬ жалюзи
000047 OUT Y6             // ВКЛЮЧИТЬ нагрев (разрешение ПИД-регулятору)
000048 TMR T11 K1200      // Запустить таймер 120 сек на открытие жалюзи + настройку ПИД

000049 LD T11             // После истечения таймера T11 (120 сек)
000050 OUT Y0             // ВКЛЮЧИТЬ вентилятор

// === БЛОК 7: ЗАЩИТА ОТ ЗАМЕРЗАНИЯ В РЕЖИМЕ НАГРЕВА ===
000051 LD M3              // Если активен режим нагрева
000052 AND X5             // И есть сигнал замерзания
000053 TMR T15 K180       // Запустить таймер 18 сек

000054 LD T15             // Если сигнал замерзания держится более 18 сек
000055 OUT M22            // Установить аварию

// === БЛОК 8: РЕЖИМ ОХЛАЖДЕНИЯ ===
000056 LD M4              // Если активен режим охлаждения
000057 ANI M22            // И нет аварий
000058 RST Y6             // Выключить нагрев
000059 OUT Y2             // ОТКРЫТЬ жалюзи
000060 TMR T12 K300       // Запустить таймер 30 сек на открытие жалюзи

000061 LD T12             // После истечения таймера T12 (30 сек)
000062 OUT Y0             // ВКЛЮЧИТЬ вентилятор
000063 TMR T20 K100       // Запустить таймер 10 сек перед включением ККБ

000064 LD T20             // После истечения таймера T20 (10 сек)
000065 OUT Y1             // ВКЛЮЧИТЬ ККБ (охлаждение)

// === БЛОК 9: КОНТРОЛЬ ДАВЛЕНИЯ ВЕНТИЛЯТОРА ===
000066 LD Y0              // Если вентилятор включен
000067 TMR T13 K300       // Запустить таймер 30 сек контроля давления

000068 LD T13             // После 30 сек работы вентилятора
000069 ANI X4             // Если нет подтверждения давления (X4=0)
000070 OUT M22            // Установить аварию

// === БЛОК 10: ИНДИКАЦИЯ СТАТУСА ===
000071 LD M22             // Если есть авария
000072 OUT Y3             // ВКЛЮЧИТЬ лампу "Авария"

000073 LD X5              // Если сигнал замерзания
000074 OUT Y4             // ВКЛЮЧИТЬ лампу "Замерз"

// === БЛОК 11: НОРМАЛЬНЫЙ ОСТАНОВ ===
000078 LD M100            // В момент остановки
000079 RST Y1             // НЕМЕДЛЕННО выключить ККБ
000080 RST Y6             // НЕМЕДЛЕННО выключить нагрев
000081 TMR T21 K120       // Задержка 120 сек (2 минуты)
000081 LD T21
000081 RST Y0             // Выключить вентилятор после 2 минут
000081 RST Y2             // Закрыть жалюзи
000081 RST M100           // Сбросить флаг

// === БЛОК 12: АВАРИЙНЫЙ ОСТАНОВ ===
000082 LD M22             // Если авария
000083 RST Y0             // НЕМЕДЛЕННО выключить вентилятор
000084 RST Y1             // НЕМЕДЛЕННО выключить ККБ
000085 RST Y6             // НЕМЕДЛЕННО выключить нагрев
000086 RST Y2             // НЕМЕДЛЕННО закрыть жалюзи

000087 END                // Конец программы
